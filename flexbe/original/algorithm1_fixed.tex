% Revised Algorithm 1 -- drop-in replacement (algorithm2e).
% Requires \usepackage[ruled,linesnumbered]{algorithm2e}
% Lines 7-11 of the published listing are preserved: R^i = bsm(I^i[0]) and
% S^i = n-k-1 on (n-m) <= k <= (n-2), else 0.  Only the index construction
% (published lines 5-6) is replaced, and the whole vector I^i is now specified.

\begin{algorithm}[t]
\SetAlgoLined
\DontPrintSemicolon
\caption{Generation of index vectors and \acs{PRS} control signals}
\label{alg:prs-control}
\KwIn{Transform size $N = 2^{n}$, number of \acsp{BU} $P_{bu}$,
      $m = \log_{2}(2P_{bu})$}
\KwOut{Index vector $\mathbf{I}^{i}$, rotation factor $R^{i}$,
       permutation state $S^{i}$}
\SetKwFunction{ins}{ins}
\tcp{$\ins(x,p) \triangleq \big((x \gg p) \ll (p{+}1)\big) \,|\, (x \bmod 2^{p})$
     inserts a zero bit at position $p$}
\BlankLine
\tcp{Butterfly stage index}
\For{$k \leftarrow 0$ \KwTo $n-1$}{
  $h \leftarrow n-1-k$ \tcp*{stage $k$ pairs indices differing in bit $h$}
  \tcp{Iterations required to complete stage $k$}
  \For{$j \leftarrow 0$ \KwTo $\tfrac{N}{2P_{bu}}-1$}{
    $i \leftarrow k \cdot \tfrac{N}{2P_{bu}} + j$ \tcp*{global iteration index}
    \eIf{$h < m$}{
      \tcp{both operands lie in \acs{RAM} row $H$}
      $H \leftarrow j$\;
      \lFor{$t \leftarrow 0$ \KwTo $P_{bu}-1$}{
        $\mathbf{I}^{i}[2t] \leftarrow (H \ll m) + \ins(t,h)$}
      $S^{i} \leftarrow h$ \tcp*{subset switch: stride permutation}
    }{
      \tcp{operands lie in rows $H$ and $H + 2^{\,h-m}$}
      $e \leftarrow j \bmod 2$\;
      $H \leftarrow \ins(\lfloor j/2 \rfloor,\; h-m)$\;
      \lFor{$t \leftarrow 0$ \KwTo $P_{bu}-1$}{
        $\mathbf{I}^{i}[2t] \leftarrow (H \ll m) + 2t + e$}
      $S^{i} \leftarrow 0$ \tcp*{subset switch: identity}
    }
    \lFor{$t \leftarrow 0$ \KwTo $P_{bu}-1$}{
      $\mathbf{I}^{i}[2t{+}1] \leftarrow \mathbf{I}^{i}[2t] + 2^{h}$}
    $R^{i} \leftarrow \mathrm{bsm}(\mathbf{I}^{i}[0])$
      \tcp*{$= \big(\mathrm{popcount}(H) + e\big) \bmod 2^{m}$}
  }
}
\end{algorithm}

% Each branch emits exactly $N/2^{m} = N/(2P_{bu})$ cycles per stage and visits
